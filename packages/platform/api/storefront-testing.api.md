```ts
import type { StorefrontVerdict } from "../../lifecycle/index.js";
/** The run subject's ledger, with the meanings of the verdict's step ledger. */
type StorefrontLedger = NonNullable<StorefrontVerdict["steps"][number]["ledger"]>;
/** A buyer action or state the scenario expects, with the actions that state must offer. */
type StorefrontExpectedProgress = {
    readonly state: string;
    readonly offers: readonly string[];
};
/**
 * A request as a command matcher sees it: the method and the request path,
 * without query string or fragment.
 */
type StorefrontMatchedRequest = {
    readonly method: string;
    readonly path: string;
};
/** `{ method, path }` matches that method and exactly that path; a predicate decides for itself. */
type StorefrontCommandMatcher = StorefrontMatchedRequest | ((request: StorefrontMatchedRequest) => boolean);
/** An anchor by its attribute group and value, or a custom CSS locator. */
type StorefrontPageTarget = {
    readonly attribute: "action" | "state" | "field" | "value";
    readonly value: string;
} | {
    readonly locator: string;
};
/** What `intercept` does with one request: let it through, deliver it and fail the page's response, or never send it. */
type StorefrontRequestDecision = "pass" | "withhold" | "block";
/**
 * One storefront proof scenario: a buyer journey with a stimulus, the ledger
 * safety it must keep, the progress it must reach and the control that must
 * keep that progress from being reached.
 *
 * Step kinds, safety rules and control kinds are run by the storefront runner.
 * This step, safety and control language is public at `testing` maturity, so
 * its shape may change in a minor release with an upgrade note.
 *
 * - `requires.outcomes` lists the outcomes a run's outcome driver must declare
 *   for the scenario to be admitted; empty means none. It names no tier: the
 *   highest tier at which a store can prove the scenario is the highest tier
 *   whose driver declares these outcomes.
 * - `budget.creates` is the number of create requests one run may send.
 * - `text` holds one sentence each on the stimulus, the safety rule, the
 *   progress and the control.
 * - Step, `noAttemptAfter`, `forbiddenStates.after` and `control.from` indices
 *   count from 0 within `steps`.
 *
 * @experimental
 */
export type StorefrontScenario = {
    readonly id: string;
    readonly title: string;
    readonly purchaseMode: "one_time" | "subscription" | (string & {});
    readonly requires: {
        readonly outcomes: readonly string[];
    };
    readonly budget: {
        readonly creates: number;
    };
    readonly text: {
        readonly stimulus: string;
        readonly safety: string;
        readonly progress: string;
        readonly control: string;
    };
    readonly steps: readonly (
    /** Store-owned steps up to the first buyer action. */
    {
        readonly kind: "arrange";
    }
    /** After a reload: arrange again, or re-type the details only; `edit` names a field to change. */
     | {
        readonly kind: "rearrange";
        readonly edit?: string;
    }
    /** Withhold the response to the first create the next `act` sends. */
     | {
        readonly kind: "withhold";
    }
    /** Perform a buyer action; `twice` performs it twice and holds the first create until the second was attempted. */
     | {
        readonly kind: "act";
        readonly action: string;
        readonly twice?: boolean;
    } | {
        readonly kind: "reload";
    }
    /** Wait for a state and the actions it must offer; `byPolicy` replaces both per changed-payload policy. */
     | (StorefrontExpectedProgress & {
        readonly kind: "expect";
        readonly byPolicy?: Readonly<Record<string, StorefrontExpectedProgress>>;
    })
    /** Ask the outcome driver for one outcome. */
     | {
        readonly kind: "outcome";
        readonly outcome: string;
    })[];
    readonly safety: {
        readonly maxPayableOrders?: number;
        readonly maxHolds?: number;
        readonly maxAttempts?: number;
        readonly maxCaptures?: number;
        /** No payment attempt after the step at this index: every later ledger read shows the attempts read when that step ended. */
        readonly noAttemptAfter?: number;
        /** Attempts never exceed the declines the outcome driver produced so far plus one. */
        readonly attemptsPerDecline?: boolean;
        /** States that must not be shown from the end of the step at index `after` on. */
        readonly forbiddenStates?: {
            readonly after: number;
            readonly states: readonly string[];
        };
    };
    readonly control: {
        readonly kind: "none" | "block_read" | "omit_action" | "other_policy" | (string & {});
        /** The state the control must keep from being reached; `null` for `none` and for `other_policy`, whose state follows the declared policy. */
        readonly negates: string | null;
        /** `block_read`: the commands blocked, from the step at index `from`. */
        readonly commands?: readonly ("create" | "read")[];
        readonly from?: number;
        /** `omit_action`: the omitted action. */
        readonly action?: string;
        /** Whether outcome steps from the control's start run (`kept`, the default) or not (`skipped`); an `omit_action` control starts at the omitted action. */
        readonly outcomes?: "kept" | "skipped";
    };
};
/**
 * A store's departures from the default proof harness. Every field is
 * optional; a field here overrides the proof file, which overrides the run's
 * defaults.
 *
 * A command matcher sees only the method and the request path, without query
 * string or fragment, so a token carried in a query never reaches a matcher or
 * a verdict. Stated limits: a store whose operations share one endpoint (for
 * example a single GraphQL path) cannot tell create from read by matcher; a
 * server-sent event stream is one request the page handle can pass or block as
 * a whole, not message by message, and WebSocket frames are not requests; requests answered by a service worker are not routed
 * unless the harness blocks service workers in its browser context; a
 * cross-origin request is matched by path alone; a redirect is followed to the
 * request the browser reports it redirected to, and a hop the browser does not
 * report is covered only by the quiet window.
 *
 * @experimental
 */
export type StorefrontAdapter = {
    /** Store-owned steps up to the first buyer action. `edit` names a field to change on a rearrangement, else `null`. */
    readonly arrange?: (context: {
        readonly scenario: StorefrontScenario;
        readonly buyer: Readonly<Record<string, string | boolean>>;
        readonly edit: string | null;
        readonly page: PageHandle;
    }) => Promise<void>;
    /** Action or state name to a CSS selector that replaces its attribute anchor. */
    readonly actions?: Readonly<Record<string, string>>;
    /** `keyHeader` names the request header that carries the command key; its value is compared, never recorded. */
    readonly commands?: {
        readonly create?: StorefrontCommandMatcher;
        readonly read?: StorefrontCommandMatcher;
        readonly keyHeader?: string;
    };
    /** Action names the UI performs by itself. */
    readonly autoProgress?: readonly string[];
    /** As the proof file's `routes`. */
    readonly routes?: {
        readonly surfaces?: Readonly<Record<string, string>>;
        readonly selectionEntry?: Readonly<Record<string, string>>;
    };
    /** Command that reruns one scenario; `{scenario}` is replaced by the scenario id. */
    readonly recheck?: string;
};
/**
 * Reads the ledger of one run subject (its e-mail): `orders` in any status,
 * `payableOrders` awaiting payment, `holds` orders holding at least one stock
 * reservation, `attempts` payment attempts, `captures` captured payments (one
 * per capture). `payableOrdersInDatabase` exists only when the observer reads a
 * database exclusive to one run; the harness must not share that database.
 *
 * @experimental
 */
export type StorefrontObserver = {
    readonly kind: "fixture" | "package" | "store" | (string & {});
    ledger(subject: string): Promise<StorefrontLedger>;
    payableOrdersInDatabase?(): Promise<number>;
};
/**
 * The page operations a storefront proof needs, over one browser page.
 * `createPageHandle` builds one from a Playwright page.
 *
 * - `path()` is the current path without query string or fragment;
 *   `width()` the viewport width in CSS pixels, or `null`.
 * - `find` scrolls the first match into view when it is not already shown in
 *   full and measures it, or resolves
 *   `null` when none appeared within `timeoutMs`; every scroll position is
 *   restored afterwards, instantly whatever the page's `scroll-behavior`. An
 *   anchor is visible when its box is non-zero; the buyer can bring it into
 *   view (judged before anything moves, from the anchor outward: a scroller the
 *   buyer can scroll brings its content into its range, while a locked one,
 *   `overflow: hidden` or `clip` on a container or on the viewport as in an app
 *   shell or a body scroll lock, must already show the anchor or an
 *   anchor-sized part of what an inner scrollable scroller shows); is not
 *   hidden (`display: none`, a `hidden` attribute up the tree, or an inherited
 *   `visibility` other than `visible`); is not fully transparent (the product of
 *   its and its ancestors' computed opacity is 0); and some part of it survives
 *   the overflow boxes of the ancestors in its containing-block chain (a fixed
 *   or absolute box escapes those outside it; the root scroller never clips)
 *   and its own and its ancestors' `clip` and `clip-path`, with the element hit at that part's centre being the anchor, a
 *   descendant or an ancestor of it (an ancestor only when no `clip-path` other
 *   than `inset()` lies on the way). Where the browser has `checkVisibility`,
 *   its answer also counts, so content of a closed `details` element or a
 *   `content-visibility: hidden` section is hidden. An anchor inside a shadow
 *   root is measured as covered (a stated limit).
 * - `anchors(locators)` lists the visible `state:<value>` and
 *   `action:<value>` anchors, sorted; `locators` maps such a name to a custom
 *   CSS locator that replaces its attribute anchor.
 * - `perform` clicks, and resolves `false` without clicking when the target is
 *   absent or disabled; `fill` types a string or sets a checkbox, `false` when
 *   absent.
 * - `intercept(decide)` routes every request through `decide`, with the request
 *   path and no query string or fragment; a `decide` that throws blocks the
 *   request. It resolves a removal function.
 * - `settled({ quietMs, timeoutMs })` resolves once every request `decide`
 *   saw has finished (its response delivered, or aborted), following a
 *   redirect to the request it was redirected to, and `quietMs` passed with no
 *   new request. It waits at most `timeoutMs` (default 30,000), then rejects
 *   with `storefront_page_settle_timeout` naming each unfinished request's
 *   method and path: an open stream (`EventSource`), a long poll or a request
 *   in flight when the page closed keeps it pending until that bound.
 *
 * @experimental
 */
export type PageHandle = {
    navigate(path: string): Promise<void>;
    reload(): Promise<void>;
    path(): string;
    width(): number | null;
    find(target: StorefrontPageTarget, options: {
        readonly timeoutMs: number;
    }): Promise<{
        width: number;
        height: number;
        insideScrollArea: boolean;
        hidden: boolean;
        transparent: boolean;
        coveredAtCentre: boolean;
    } | null>;
    anchors(locators?: Readonly<Record<string, string>>): Promise<string[]>;
    perform(target: StorefrontPageTarget, options: {
        readonly timeoutMs: number;
    }): Promise<boolean>;
    fill(target: StorefrontPageTarget, value: string | boolean, options: {
        readonly timeoutMs: number;
    }): Promise<boolean>;
    intercept(decide: (request: StorefrontMatchedRequest & {
        header(name: string): string | null;
    }) => StorefrontRequestDecision | Promise<StorefrontRequestDecision>): Promise<() => Promise<void>>;
    settled(options: {
        readonly quietMs: number;
        readonly timeoutMs?: number;
    }): Promise<void>;
};
export {};
```

```ts
import { type StorefrontProofFile, type StorefrontVerdict } from "../../lifecycle/index.js";
import type { PageHandle, StorefrontAdapter, StorefrontObserver, StorefrontScenario } from "./contracts.js";
type CommandMatcher = NonNullable<NonNullable<StorefrontAdapter["commands"]>["create"]>;
/**
 * Options of one scenario run. `page` is a `PageHandle` (from
 * `createPageHandle`); `commands` are the host proof kit's create and read
 * matchers and the command key header (default `idempotency-key`), each
 * overridden by the adapter's. `width` is required for `browser_ui` and must
 * equal `page.width()` when that is known. `clock.now()` returns integer epoch
 * milliseconds; it times the steps and gives `completedAt`. `runId` matches
 * `^[A-Za-z0-9-]{8,40}$`; the run subjects are
 * `storefront-<runId>-clean@example.test` and `…-control@example.test`.
 * `timeoutMs` (default 10,000) bounds each anchor wait; `quietMs` (default
 * 3,000) is the quiet window that ends every step. An observer with
 * `payableOrdersInDatabase` declares that its database is exclusive to this
 * run; the harness must not share it. `outcomes` is the outcome driver:
 * `capabilities` declares the outcome kinds it can produce, `produce` produces
 * one for the run's subject. `stopAt` (`checkout.<state>` for a state the
 * scenario expects) drives the clean run until that state is visible.
 */
type StorefrontRunOptions = {
    readonly scenario: string | StorefrontScenario;
    readonly page: PageHandle;
    readonly observer: StorefrontObserver;
    readonly tier: StorefrontVerdict["tier"];
    readonly profile: string;
    readonly width?: number;
    readonly channel?: StorefrontVerdict["channel"];
    readonly commands?: {
        readonly create?: CommandMatcher;
        readonly read?: CommandMatcher;
        readonly keyHeader?: string;
    };
    readonly proofFile?: StorefrontProofFile;
    readonly adapter?: StorefrontAdapter;
    readonly build: Omit<StorefrontVerdict["build"], "source"> & {
        readonly source?: StorefrontVerdict["build"]["source"];
    };
    readonly harness: {
        readonly runnerVersion: string;
        readonly adapterDigest?: string;
    };
    readonly runId: string;
    readonly clock: {
        now(): number;
    };
    readonly timeoutMs?: number;
    readonly quietMs?: number;
    readonly outcomes?: {
        readonly capabilities: readonly string[];
        produce(outcome: string, context: {
            readonly subject: string;
        }): Promise<void>;
    };
    readonly stopAt?: string;
};
/** A full run's verdict, or with `stopAt` the clean run's steps up to the state it stopped at. */
type StorefrontRunResult = {
    kind: "verdict";
    verdict: StorefrontVerdict;
} | {
    kind: "stopped";
    stoppedAt: string;
    steps: StorefrontVerdict["steps"];
};
/**
 * Runs one storefront scenario and resolves `{ kind: "verdict", verdict }`, a
 * format-1 verdict that `parseStorefrontVerdict` accepts, or with `stopAt`
 * `{ kind: "stopped", stoppedAt, steps }`.
 *
 * The clean run reads the subject's ledger (an earlier order ends `not_run`),
 * installs one interceptor that classifies requests by the create and read
 * matchers, records each command's key reuse (key values are compared in
 * memory, never stored) and applies the armed withhold or hold, then runs the
 * steps. Every step ends only when the page has settled: every request the
 * interceptor saw finished and `quietMs` passed with no new request. Then the
 * runner reads the ledger, the visible anchors and the path. After the last
 * step a settle window waits once more and reads them again. The control run
 * repeats the steps under the other subject with the scenario's control: a
 * `block_read` control blocks every request of its commands from its step; an
 * `omit_action` control omits its action and, when the scenario says so, the
 * outcomes after it; an `other_policy` control changes nothing and must not
 * show the progress of the changed-payload policy the proof file does not
 * declare. A `safety_only` scenario has no control run. When the UI performs
 * the omitted action by itself, as the effective `autoProgress` declares,
 * neither run performs it and the control becomes a `block_read` of status
 * reads from the step before it, with the outcomes after it skipped.
 *
 * Outcomes: an `outcome` step asks the `outcomes` driver for one outcome
 * (`declined`, `pending`, `settled`, `settled_after_pending`,
 * `activation_pending`). A scenario runs only when every outcome in its
 * `requires.outcomes` is one this runner knows and the driver declares; else
 * it ends `not_run` with `tier_unreachable` naming each missing outcome before
 * any page, observer or driver call. Tier names do not decide admission. A
 * rejected `produce`, or a ledger without a payment attempt after the outcome,
 * is `outcome_failed`.
 *
 * Judgement, first failing group decides: 0 admitted; 1 page, observer,
 * arrangement and outcome driver worked; 2 no order before the arrangement and
 * one after the first matched create; 3 the first order appeared in a step with
 * a matched create, every withhold applied and every outcome shows an attempt;
 * 4 ledger safety in both runs and settle windows (payable orders, orders
 * holding stock, attempts, captures, no attempt after its step, attempts per
 * decline, forbidden states from their step on; a paid state needs a capture
 * and no payable order) and, for an exclusive database, no payable order
 * outside the subjects; 5 the create budget kept (a run over it stopped before
 * proving anything, but a violation it already caused stands); 6 each matcher
 * used matched and each needed anchor resolved and was visible; 7 the clean run
 * reached its expected state (the other policy's state is `policy_mismatch`);
 * 8 a `block_read` control blocked a request and the negated state was not
 * reached, judged over every recorded control step even when the control run
 * stopped; under an `omit_action` control a negated state reached after a
 * status read is `auto_progress_undeclared`. Codes 0–3, 5 and 6 end `not_run`,
 * as do `control_not_applied` and `auto_progress_undeclared`; a harness fault
 * is never `failed` or `passed`; a ledger violation is
 * `failed`; progress under the control is `vacuous`. Without a finding the
 * result is `passed`, or `safety_only` for a scenario without a control.
 *
 * Invalid options (an unknown scenario, a malformed `runId`, a missing or
 * mismatched `width`, a clock outside integer milliseconds 0 to the end of the
 * year 9999, a malformed outcome driver) reject with a `TypeError` starting
 * `storefront_scenario_unknown` or `storefront_run_input_invalid`, and nothing
 * is driven, as does a malformed scenario object; a `stopAt` that is not
 * `checkout.<state>` for a state the scenario expects rejects
 * `storefront_stop_state_unknown`. With `stopAt` the runner runs the clean
 * steps until that state is visible, removes its interceptor and leaves the
 * page there: no control run and no verdict; a state not reached rejects
 * `storefront_stop_state_unreached` naming the last step, as does a scenario
 * that is not admitted, before any call. An arrangement that
 * fails under a control is the control's effect: the remaining steps are
 * skipped and the settle window still samples the page. Anchors and the ledger
 * are sampled after each step settles and in the settle window, so a state
 * shown only inside a step is not seen. A fault's text keeps its code and
 * first sentence with every URL replaced. The clean and control runs share the
 * page and its browser storage.
 *
 * @experimental
 */
export declare function runStorefrontScenario(options: StorefrontRunOptions): Promise<StorefrontRunResult>;
export {};
```

```ts
import type { StorefrontScenario } from "./contracts.js";
/**
 * The storefront proof scenarios this runner knows, frozen. Scenario ids are
 * read by stores and reports and are never renamed; a wrong scenario is
 * superseded by a new id. `runStorefrontScenario` runs one of them by id.
 *
 * - `lost_response_reload`: the response to the first place-order request is
 *   lost after the server received it; the buyer reloads and places the order
 *   again. At most one payable order, one order holding stock and one capture;
 *   progress is `awaiting_payment` offering `continue_payment`; the control
 *   blocks status reads and creates from the reload on and must keep
 *   `awaiting_payment` from being shown.
 * - `duplicate_submit`: the buyer places the order twice before the first
 *   request is answered. At most one payable order, one order holding stock,
 *   one payment attempt and one capture; no progress is judged and there is no
 *   control run, so a clean run ends `safety_only`.
 * - `lost_response_edited_payload`: as `lost_response_reload`, but the buyer
 *   changes the first name before placing the order again. Progress follows the
 *   proof file's `changedPayloadPolicy`: `update_in_place` shows
 *   `awaiting_payment` offering `continue_payment`, `refuse_changed` (also when
 *   absent or unknown) shows `details_changed` offering `continue_payment` and
 *   `start_over`. The `other_policy` control runs the steps again and must not
 *   show the other policy's progress.
 * - `reload_pending_attempt`: a payment attempt left pending is reloaded, then
 *   settled. No new payment attempt from the reload on and at most one capture;
 *   progress is `paid`; the control blocks status reads and creates from the
 *   reload on, keeps the outcomes and must keep `paid` from being shown (it may
 *   still show `awaiting_authority`). Requires the `pending` and
 *   `settled_after_pending` outcomes.
 * - `decline_retry`: a declined payment is retried and settled. At most one
 *   payable order, attempts never more than declines plus one, at most one
 *   capture; progress is `paid`; the `omit_action` control omits
 *   `retry_payment`, skips the outcomes after it and must keep `paid` from
 *   being shown. Requires the `declined` and `settled` outcomes.
 * - `paid_pending_activation`: a subscription's payment settles while its
 *   activation stays pending. From the outcome on, `awaiting_payment`,
 *   `declined` and `paid` must not be shown, and at most one capture; progress
 *   is `paid_pending_activation`; the control blocks status reads and creates
 *   from the outcome on, keeps the outcomes and must keep that state from being
 *   shown. Requires the `activation_pending` outcome.
 *
 * @experimental
 */
export declare const storefrontScenarios: readonly StorefrontScenario[];
```

```ts
 * UTF-16 code units, and a line feed, after `parseStorefrontVerdict` accepts

 * observation, then `Cause:`, `Fix:` and `Recheck:`. A harness writes one file

 * file.
```
