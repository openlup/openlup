```ts
/**
 * A per-job coordination capability. Its exclusion scope and serialization behavior
 * depend on the implementation; hosts requiring cross-instance exclusion must
 * supply a cluster-scoped lock.
 */
export interface AdvisoryLock {
    /** Explicit capability marker; a host that must not run a job twice requires a cluster lock. */
    readonly durability?: "cluster" | "process";
    /** Try to acquire the lock for `jobId` without blocking. */
    tryAcquire(jobId: string): Promise<boolean>;
    /** Release a previously-acquired lock for `jobId`. */
    release(jobId: string): Promise<void>;
    /** Release the dedicated session when the host stops; safe to call repeatedly. */
    close(): Promise<void>;
}
/** Minimal query surface; the caller must supply a session-affine client, never Pool.query. */
export interface AdvisoryLockQuerier {
    query<T = unknown>(text: string, values?: unknown[]): Promise<{
        rows: T[];
    }>;
    end?(): Promise<void>;
}
/**
 * Deterministic signed-32-bit key from a jobId for `pg_try_advisory_lock(bigint)`.
 * The hash is stable across processes, but distinct job ids can collide and share
 * a lock key.
 */
export declare function advisoryLockKey(jobId: string): number;
/**
 * pg-backed advisory lock. Uses `pg_try_advisory_lock` (session-scoped, non-blocking) on acquire
 * and `pg_advisory_unlock` on release. NOTE: session-scoped locks require the same connection for
 * acquire+release, so the supplied querier should be a single dedicated client/connection, not a
 * pool that hands out a different connection per query. Each grant is checked in a separate
 * statement against the backend PID and pg_locks before protected work is admitted. This detects
 * observed transport violations; it cannot certify a pooler's future routing configuration.
 */
export declare function createPgAdvisoryLock(client: AdvisoryLockQuerier, closeClient?: () => Promise<void>): AdvisoryLock;
/**
 * Build the lock a long-running host should actually use, from the connection string the
 * environment declares. Returns null when no connection string is configured, which leaves the
 * scheduler on its always-granting default — the exact behaviour of a host with a single instance.
 *
 * The connection is opened LAZILY, on the first acquire, and then reused: a session-scoped
 * try-lock must acquire and release on the SAME connection, and a host that never fires a timer
 * must never open one. `pg` stays out of the static import graph (dynamic import), so the default
 * hosted path and typecheck are untouched.
 */
export interface EnvAdvisoryLockOptions {
    /** Optional sink for dedicated pg.Client `'error'` events; defaults to a non-throwing stderr logger. */
    onClientError?: (error: Error) => void;
}
export declare function createEnvAdvisoryLock(env: Record<string, string | undefined>, connect?: (connectionString: string) => Promise<AdvisoryLockQuerier>, options?: EnvAdvisoryLockOptions): AdvisoryLock | null;
/** Lock that always grants (single-instance / tests). Never use in multi-instance Node deploys. */
export declare function createNoopAdvisoryLock(): AdvisoryLock;
```

```ts
     * The declared HTTP entrypoint (Vercel cron invokes it directly). A Node

    crons?: Array<{
        path: string;
```

## dist/scheduler/nodeScheduler.d.ts

```ts
import type { SchedulerPort } from "@openlup/core/platform-runtime";
import type { ScheduledJobDeclaration } from "./jobRegistry.js";
import { type AdvisoryLock } from "./advisoryLock.js";
type CronEnv = Record<string, string | undefined> & {
    CRON_SECRET?: string;
};
/** Minimal node-cron surface we depend on (keeps the package optional at type level). */
export interface CronScheduler {
    schedule(expression: string, handler: () => void | Promise<void>, options?: {
        timezone?: string;
    }): {
        stop: () => void;
    };
}
export interface NodeSchedulerAdapter extends SchedulerPort {
    readonly jobs: readonly ScheduledJobDeclaration[];
    /** Fence/stop current timers synchronously, then await admitted callbacks and lock release. Later registration remains supported. */
    stopAll(): Promise<void>;
}
export type SchedulerLockMode = "cluster" | "process";
export interface CreateNodeSchedulerOptions {
    /** node-cron instance (inject `import("node-cron")` at the host boot site). */
    cron: CronScheduler;
    /**
     * Per-job lock capability; defaults to an always-granting no-op for
     * single-instance/test use when `lockMode` is `"process"`, which does not serialize concurrent callbacks.
     * Supply a cluster-scoped lock (`durability: "cluster"`) when `lockMode` is `"cluster"`.
     */
    lock?: AdvisoryLock;
    /**
     * Lock enforcement mode: `"process"` (default for direct 1.x callers) falls back to a no-op lock
     * when `lock` is omitted; `"cluster"` fails closed unless `lock?.durability === "cluster"`.
     */
    lockMode?: SchedulerLockMode;
    /** Environment whose CRON_SECRET verifies an invocation. */
    env: CronEnv;
    /** The deployment's declared jobs (see `loadJobRegistry`). */
    jobs: readonly ScheduledJobDeclaration[];
    /** Timezone node-cron uses to interpret the UTC-authored schedule; pass `UTC` for Vercel parity. */
    timezone?: string;
    /** Structured error sink for failed job bodies (defaults to console.error). */
    onError?: (jobId: string, error: unknown) => void;
}
/**
 * Wrap a job body with the supplied per-job lock capability. Returns a guarded runner that:
 *  - tries to acquire the lock; if acquisition is declined, skips without running the body;
 *  - runs the body; always releases the lock in finally.
 */
export declare function withAdvisoryLock(jobId: string, lock: AdvisoryLock, body: () => Promise<void> | void, onError: (jobId: string, error: unknown) => void): () => Promise<void>;
/** Build the Node scheduler adapter backed by node-cron and the supplied lock capability. */
export declare function createNodeScheduler(options: CreateNodeSchedulerOptions): NodeSchedulerAdapter;
export {};
```

```ts
import type { PlatformHttpRequest, SchedulerPort } from "@openlup/core/platform-runtime";
import type { ScheduledJobDeclaration } from "./jobRegistry.js";
type CronEnv = Record<string, string | undefined> & {
    CRON_SECRET?: string;
};
export { bearerToken } from "./cronSecret.js";
/**
 * Verify a cron invocation is authentic. Fail-CLOSED:
 *  - CRON_SECRET unset                         -> false (never authorize when no secret configured)
 *  - presented Bearer token != CRON_SECRET     -> false
 *  - presented Bearer token == CRON_SECRET     -> true
 * This is the single canonical check that replaces the per-handler inline comparisons.
 */
export declare function verifyCronAuth(req: PlatformHttpRequest, env: CronEnv): boolean;
export interface VercelSchedulerAdapter extends SchedulerPort {
    /** The jobs this scheduler knows about, derived from the platform-runtime single source. */
    readonly jobs: readonly ScheduledJobDeclaration[];
}
/**
 * Build the Vercel cron scheduler adapter.
 *
 * `register()` validates the job is part of the declared registry (so callers can't silently
 * register a job Vercel will never fire) but does NOT start a timer — Vercel's cron does that.
 */
export declare function createVercelScheduler(options: {
    env: CronEnv;
    /** The deployment's declared jobs (see `loadJobRegistry`). */
    jobs: readonly ScheduledJobDeclaration[];
}): VercelSchedulerAdapter;
```
