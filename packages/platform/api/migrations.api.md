```ts
import type { Pool, PoolClient } from "pg";
import type { MigrationRunnerPort } from "@openlup/core/platform-runtime";
export interface PostgresMigrationRunnerEnv {
    connectionString: string;
}
/** One migration file of the deployment's platform catalogue. */
export type PlatformMigrationFile = {
    file: string;
    content: string;
};
export type PlatformMigrationManifestEntry = {
    file: string;
    sha256: string;
};
/** The validated manifest: a baseline, its ordered forwards and the expected object inventory. */
export type PlatformMigrationManifest = {
    schemaVersion: 1;
    baseline: PlatformMigrationManifestEntry;
    forward: PlatformMigrationManifestEntry[];
    objectInventorySha256: string;
};
/**
 * The deployment's migration catalogue and its validator. `evaluate` returns every refusal for a
 * manifest against the catalogue files and, when given, the live object-inventory payload.
 */
export interface PlatformMigrationSource {
    readMigrations(root: string): {
        migrations: PlatformMigrationFile[];
        errors: string[];
    };
    readManifest(root: string): unknown | null;
    evaluate(input: {
        manifest: unknown;
        migrations: readonly PlatformMigrationFile[];
        objectInventoryPayload?: string;
    }): string[];
    /**
     * Optional trusted-source observation/refusal after exact-prefix validation, under the
     * migration session lock. Runs once per operation before any pending SQL or dry-run result.
     * The canonical source uses read-only observations; the callback shares the caller's existing
     * database capability and is not a sandbox or a provisioning/per-unit execution hook.
     */
    assertSession?(client: PoolClient, applied: readonly (PlatformMigrationManifestEntry & {
        position: number;
    })[]): Promise<void>;
}
export interface PostgresMigrationRunnerOptions {
    /** Name of the PostgreSQL advisory lock that serializes migration sessions. */
    lockName: string;
    /** The deployment's migration catalogue. */
    source: PlatformMigrationSource;
    /** Serializes the live object inventory that the manifest's `objectInventorySha256` pins. */
    catalogProjectionPayload: (client: PoolClient) => Promise<string>;
    /** Catalogue root; defaults to process.cwd(). */
    root?: string;
    poolFactory?: (env: PostgresMigrationRunnerEnv) => Pool | Promise<Pool>;
}
export declare function createPostgresMigrationRunner(env: PostgresMigrationRunnerEnv, options: PostgresMigrationRunnerOptions): MigrationRunnerPort;
export declare function resolvePostgresMigrationRunnerEnv(env: Record<string, string | undefined>): PostgresMigrationRunnerEnv;
```

```ts
import type { ClientBase } from "pg";
type R<T> = Readonly<T>;
type Sha256 = string & {
    readonly __sha256: unique symbol;
};
export type CatalogKind = "function" | "relation" | "column" | "index" | "constraint" | "trigger" | "policy" | "acl-grant" | "default-privilege" | "role-membership" | "type" | "schema" | "extension" | "role";
export type ObjectIdentity = R<{
    kind: CatalogKind;
    schema: string | null;
    name: string;
    arguments: readonly string[];
}>;
export type AclTargetClass = "schema" | "routine" | "relation" | "sequence" | "column" | "type";
export type AclGrant = R<{
    identity: ObjectIdentity;
    targetClass: AclTargetClass;
    targetIdentity: ObjectIdentity;
    grantor: string;
    grantee: string;
    privilege: string;
    isGrantable: boolean;
    aclOrigin: "stored" | "built-in-default";
}>;
export type DefaultPrivilege = R<{
    identity: ObjectIdentity;
    ownerRole: string;
    schema: string | null;
    objectClass: "schema" | "routine" | "relation" | "sequence" | "type";
    grantor: string;
    grantee: string;
    privilege: string;
    isGrantable: boolean;
}>;
export type RoleMembership = R<{
    identity: ObjectIdentity;
    grantedRole: string;
    memberRole: string;
    grantorRole: string;
    adminOption: boolean;
    inheritOption: boolean;
    setOption: boolean;
}>;
export type RoleRecord = R<{
    identity: ObjectIdentity;
    canLogin: boolean;
    inherits: boolean;
    superuser: boolean;
    createRole: boolean;
    createDb: boolean;
    replication: boolean;
    bypassRls: boolean;
    connectionLimit: number;
    validUntil: string | null;
}>;
export type ObjectOwnership = R<{
    object: ObjectIdentity;
    ownerRole: string;
}>;
export type ProjectedObject = R<{
    identity: ObjectIdentity;
    definitionDigest: Sha256 | null;
    attributesDigest: Sha256;
}>;
declare const validatedProjection: unique symbol;
export type ValidatedProjection = R<{

    roleMemberships: readonly RoleMembership[];
    roles: readonly RoleRecord[];
    ownerships: readonly ObjectOwnership[];
    digest: Sha256;
    readonly [validatedProjection]: "selected-catalog-v2";
}>;
type CatalogRow = {
    kind: CatalogKind;
    schema: string | null;
    name: string;
    arguments: string[];
    definition: string | null;
    attributes: string;
};
```
