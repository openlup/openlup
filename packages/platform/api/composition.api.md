import type { HttpRequest, HttpResponse } from "../http/index.js";
export type RouteEntry = {
    route: string;
    pattern: RegExp;
    params: string[];
    handler: (req: HttpRequest, res: HttpResponse) => unknown;
    /** Fail unsupported hosted bundles closed before auth without weakening direct-route auth. */
    availability?: "direct-postgres";
};
/** Match `routes` against one request and run the handler, or answer not found. */
export type BffDispatch = (routes: readonly RouteEntry[], req: HttpRequest, res: HttpResponse) => Promise<unknown>;
export interface BffDispatcherOptions {
    /** Environment read on each request to decide a route's bundle availability. */
    env: () => Record<string, string | undefined>;
    /** Writes the deployment's not-found response for an unmatched or unavailable path. */
    notFound: (res: HttpResponse, pathname: string) => void;
    /** Runs once per request before matching (for example, ambient request configuration). */
    beforeDispatch?: () => void;
    /**
     * Runs after matching and before the handler, for example route authorization; resolving false
     * means it already answered. Required, so no composition skips it by omission; a deployment
     * without guarded routes passes `() => true`.
     */
    preflight: (req: HttpRequest, res: HttpResponse, route: string) => Promise<boolean> | boolean;
}
/** Build the dispatcher a deployment mounts on its hosted and Node entrypoints alike. */
export declare function createBffDispatcher(options: BffDispatcherOptions): BffDispatch;

import type { AnalyticsPort, BlobStoragePort, DataGatewayPort, HttpRuntimePort, MigrationRunnerPort, SchedulerPort, TransactionalRuntimePort } from "@openlup/core/platform-runtime";
import { type PlatformBundleDescriptor, type PlatformBundleId } from "../kernel/index.js";
interface PaymentServices {
    run<T>(work: (port: unknown) => Promise<T>): Promise<T>;
}
type Env = Record<string, string | undefined>;
export interface PlatformBundlePorts {
    fulfillment?: {
        runDispatch(): Promise<unknown>;
    } | null;
    payments?: PaymentServices | null;
    http: HttpRuntimePort | null;
    scheduler: SchedulerPort | null;
    blob: BlobStoragePort | null;
    data: DataGatewayPort | null;
    migrations: MigrationRunnerPort | null;
    analytics: AnalyticsPort | null;
    transactional: TransactionalRuntimePort | null;
}
export interface PlatformBundle {
    id: PlatformBundleId;
    descriptor: PlatformBundleDescriptor;
    ports: PlatformBundlePorts;
}
/**
 * One binder per port slot. Each receives the slot's capability and the environment; the blob
 * binder receives the selected blob capability (`PLATFORM_BLOB`, else the bundle's own), and the
 * transactional binder also receives the bundle's data capability.
 */
export interface PlatformBundleBinders {
    /** Explicit host selection only; no default binder. */
    fulfillment?: (env: Env) => {
        runDispatch(): Promise<unknown>;

    payments?: (env: Env) => PaymentServices | null;

     * structurally here so the composition entry's declarations reference no data-module type.

        auth?: {
            persistSession?: boolean;
            autoRefreshToken?: boolean;
        };
    }) => unknown;
    http?: (capability: string, env: Env) => HttpRuntimePort | null;
    scheduler?: (capability: string, env: Env) => SchedulerPort | null;
    /** Defaults to `bindBlobPort`. */
    blob?: (capability: string, env: Env) => BlobStoragePort | null;
    /** For example `createDataBindings(...).bindDataPort`. */
    data?: (capability: string, env: Env) => DataGatewayPort | null;
    migrations?: (capability: string, env: Env) => MigrationRunnerPort | null;
    analytics?: (capability: string, env: Env) => AnalyticsPort | null;
    transactional?: (capability: string, dataCapability: string, env: Env) => TransactionalRuntimePort | null;
}
export type ComposeBundle = (env: Env) => PlatformBundle;

import type { HttpRuntimePort, JobLeasePort } from "@openlup/core/platform-runtime";
import type { StaticAssetReader } from "../http/index.js";
import { type AdvisoryLock, type CronScheduler, type ScheduledJob, type ScheduledJobDeclaration, type ScheduledJobHandler, type ScheduledJobHandlerRegistry } from "../scheduler/index.js";
import type { ComposeBundle, PlatformBundlePorts } from "./composeBundle.js";
type Env = Record<string, string | undefined>;
/** What the host hands its admission check before it serves or schedules anything. */
export interface NodeHostAdmission {
    kind: "node";
    nodeJobHandlers: ScheduledJobHandlerRegistry;
    /** The path-declared jobs this process will time; empty when it does not run the scheduler. */
    nodeJobs: readonly ScheduledJobDeclaration[];
    /** The registry jobs this process will run directly; empty when it does not run the scheduler. */
    scheduledJobs: readonly ScheduledJob[];
}
export interface NodeHostDependencies {
    /** The deployment's composition root (see `createBundleComposer`). */
    composeBundle: ComposeBundle;
    /**
     * Build the HTTP runtime this host serves, for the declared jobs and its static reader. The
     * bundle's `http` port only proves that the bundle is a Node bundle at readiness time.
     */
    createHttpRuntime: (input: {
        schedulerJobs: readonly ScheduledJobDeclaration[];
        staticReader: StaticAssetReader;
        /** The composed ports (blob storage, data gateway and the rest) for routes to use. */
        ports: PlatformBundlePorts;
    }) => HttpRuntimePort;
    /** The deployment's declared jobs, used when the caller passes none. */
    loadSchedulerJobs: () => readonly ScheduledJobDeclaration[];
    /** Transport-neutral jobs contributed to the registry; run in-process with no HTTP secret. */
    scheduledJobs?: (env: Env, ports: PlatformBundlePorts) => readonly ScheduledJob[];
    /** The job-lease port for registry jobs that declare a lease. */
    jobLease?: (env: Env) => JobLeasePort;
    /** Direct bindings the deployment and its domains contribute; caller overrides win. */
    scheduledJobHandlers?: (env: Env) => ScheduledJobHandlerRegistry;
    /** Runs first, before readiness (for example, process-wide configuration checks). */
    prepareHost?: (env: Env) => void;
    /** Admits the host before it serves or schedules; a rejection stops the boot. */
    admitHost?: (env: Env, admission: NodeHostAdmission) => Promise<void>;
    /** Operational log sink (defaults to console.log). */
    log?: (message: string) => void;
}
/**
 * A deployment's scheduler profile: a fenced projection of the declared jobs with its own
 * validation, applied at the host's boot boundaries. Each hook may throw to stop the boot.
 */
export interface NodeSchedulerProfile {
    /** Replaces the declared jobs with this profile's projection. Runs after any migration replay. */
    projectJobs(env: Env, jobs: readonly ScheduledJobDeclaration[]): readonly ScheduledJobDeclaration[];
    /** Direct bindings the profile adds; they replace the deployment's and the caller's for a job. */
    handlers?(env: Env): ScheduledJobHandlerRegistry;
    /** Runs once the advisory lock is resolved and before any job is registered. */
    validateSchedule?(input: {
        lock: AdvisoryLock | undefined;
        jobs: readonly ScheduledJobDeclaration[];
        overrides: ScheduledJobHandlerRegistry | undefined;
    }): void;
    /**
     * Runs for each job, in order, immediately before it is registered, with its direct binding if
     * any. A refusal stops the boot; jobs registered before it keep their timers, as the boot fails.
     */
    validateJob?(job: ScheduledJobDeclaration, directHandler: ScheduledJobHandler | undefined): void;
}
export interface ServeNodeOptions {
    /** The process environment of this host. */
    env: Env;
    /** dist/ root for static serving (defaults to <cwd>/dist). */
    distDir?: string;
    /** node-cron instance; injected so the dependency stays optional for typecheck and tests. */
    cron?: CronScheduler;
    /** Direct bindings added over the deployment's own for this boot. */
    scheduledJobHandlers?: ScheduledJobHandlerRegistry;
    /** Explicit scheduler projection; production uses the deployment's job registry. */
    schedulerJobs?: readonly ScheduledJobDeclaration[];
    /** Optional scheduler profile applied to the declared jobs (see `NodeSchedulerProfile`). */
    schedulerProfile?: NodeSchedulerProfile;
    /**
     * Per-job cluster mutex. Defaults to the one the environment's connection string describes, so a
     * host that runs more than one instance stops executing every timer once per instance.
     */
    advisoryLock?: AdvisoryLock;
    /** Skip binding the network listener (unit tests build the runtime without opening a socket). */
    listen?: boolean;
    port?: number;
    host?: string;
}
export interface ServeNodeResult {
    bundleId: string;
    httpRuntime: HttpRuntimePort;
    registeredJobs: string[];
    /** Fence timers, drain admitted jobs, then close the local listener and lock session. Safe to call repeatedly. */
    stop(): Promise<void>;
}
export interface NodeHost {
    /** Readiness gate: the active bundle's descriptor decides, and the bundle must bind a Node runtime. */
    assertBundleReady(env: Env): {
        bundleId: string;
    };
    /** Build (and optionally start) the Node host. */
    serveNode(options: ServeNodeOptions): Promise<ServeNodeResult>;
}
export declare function createNodeHost(deps: NodeHostDependencies): NodeHost;
export {};

## dist/scheduler/advisoryLock.d.ts

```ts
/**
 * A per-job coordination capability. Its exclusion scope and serialization behavior
 * depend on the implementation; hosts requiring cross-instance exclusion must
 * supply a cluster-scoped lock.
 */
export interface AdvisoryLock {
    /** Explicit capability marker; a host that must not run a job twice requires a cluster lock. */
    readonly durability?: "cluster" | "process";
    /** Try to acquire the lock for `jobId` without blocking. */
    tryAcquire(jobId: string): Promise<boolean>;
    /** Release a previously-acquired lock for `jobId`. */
    release(jobId: string): Promise<void>;
    /** Release the dedicated session when the host stops; safe to call repeatedly. */
    close(): Promise<void>;
}
/** Minimal query surface; the caller must supply a session-affine client, never Pool.query. */
export interface AdvisoryLockQuerier {
    query<T = unknown>(text: string, values?: unknown[]): Promise<{
        rows: T[];
    }>;
    end?(): Promise<void>;
}
/**
 * Deterministic signed-32-bit key from a jobId for `pg_try_advisory_lock(bigint)`.
 * The hash is stable across processes, but distinct job ids can collide and share
 * a lock key.
 */
export declare function advisoryLockKey(jobId: string): number;
/**
 * pg-backed advisory lock. Uses `pg_try_advisory_lock` (session-scoped, non-blocking) on acquire
 * and `pg_advisory_unlock` on release. NOTE: session-scoped locks require the same connection for
 * acquire+release, so the supplied querier should be a single dedicated client/connection, not a
 * pool that hands out a different connection per query. Each grant is checked in a separate
 * statement against the backend PID and pg_locks before protected work is admitted. This detects
 * observed transport violations; it cannot certify a pooler's future routing configuration.
 */
export declare function createPgAdvisoryLock(client: AdvisoryLockQuerier, closeClient?: () => Promise<void>): AdvisoryLock;
/**
 * Build the lock a long-running host should actually use, from the connection string the
 * environment declares. Returns null when no connection string is configured, which leaves the
 * scheduler on its always-granting default — the exact behaviour of a host with a single instance.
 *
 * The connection is opened LAZILY, on the first acquire, and then reused: a session-scoped
 * try-lock must acquire and release on the SAME connection, and a host that never fires a timer
 * must never open one. `pg` stays out of the static import graph (dynamic import), so the default
 * hosted path and typecheck are untouched.
 */
export declare function createEnvAdvisoryLock(env: Record<string, string | undefined>, connect?: (connectionString: string) => Promise<AdvisoryLockQuerier>): AdvisoryLock | null;
/** Lock that always grants (single-instance / tests). Never use in multi-instance Node deploys. */
export declare function createNoopAdvisoryLock(): AdvisoryLock;
```

    env: CronEnv;
