import type { PaymentExecutionAttemptStatus, PaymentFailureClass, PaymentFailureHint, PaymentProviderCapabilityDescriptor, PaymentRetryAdviceCode, StoredMandateSnapshot } from "@openlup/core/payment";
import type { PaymentAbsencePort, PaymentReadbackPort, PaymentVerifyNormalizePort } from "./evidence.js";
declare const transientSecret: unique symbol;
/**
 * One-time customer authorization or opaque client handoff, held only in memory.
 * Never persist, log, fingerprint, or include in request/response diagnostics.
 * The host brands a value only at the transient client/server input boundary.
 */
export type TransientSecret = string & {
    readonly [transientSecret]: true;
};
/** Hosted navigation or client confirmation; opaque payloads are transient only. */
export type PaymentMethodHandoff = {
    kind: "redirect";
    url: string;
} | {
    kind: "client-confirmed";
    payload: TransientSecret;
};
/** Only core neutral taxonomy crosses the adapter boundary; no provider prose. */
export interface PaymentRefusalEnvelope {
    code: PaymentFailureClass;
    hints: readonly PaymentFailureHint[];
    adviceCode?: PaymentRetryAdviceCode;
}
/**
 * A non-terminal attempt or synchronous decline; terminal money is evidence-owned.
 * Diagnostics must exclude transient secrets, handoff payloads, raw bodies and
 * provider prose. Persist only explicitly redacted request/response diagnostics.
 */
export interface PaymentExecutionEnvelope {
    outcome: {
        kind: "attempt";
        status: PaymentExecutionAttemptStatus;
    } | {
        kind: "decline";
        refusal: PaymentRefusalEnvelope;
    };
    dispatch: {
        kind: "dispatched";
        providerPaymentRef: string;
    } | {
        kind: "notDispatched";
        evidence: {
            exact: true;
            phase: "before-dispatch";
            reason: PaymentRefusalEnvelope;
        };
    } | {
        kind: "unknown";
    };
    handoff?: PaymentMethodHandoff;
    diagnostics: Readonly<Record<string, string | number | boolean | null>>;
}
/** Host identity/key/fingerprint and configured integer-minor money pass verbatim. */
export interface PaymentAttemptPort {
    validateInput(input: {
        attemptRef: string;
        intentRef: string;
        fingerprint: string;
        providerIdempotencyKey: string;
        money: {
            amountMinor: number;
            currency: string;
        };
        mode: "live" | "test";
        purpose: "initial" | "renewal-unattended" | "recovery";
        storedMethod?: {
            reference: string;
            mandate: StoredMandateSnapshot;
        };
        /** Client-entered, server-submitted authorization; never persisted or echoed. */
        customerAuthorization?: TransientSecret;
    }): void;
    execute(input: Parameters<PaymentAttemptPort["validateInput"]>[0]): Promise<PaymentExecutionEnvelope>;
}
/** Platform overlay; the shared core descriptor remains unchanged. */
export interface PaymentAdapterCapabilities extends Omit<PaymentProviderCapabilityDescriptor, "captureFlows"> {
    captureFlows: readonly {
        kind: string;
        handoff: PaymentMethodHandoff["kind"] | "customer-authorization";
        methodFamilies: readonly ("card" | "wallet" | "bank-redirect" | "one-time-code")[];
    }[];
    supportedCurrencies: readonly string[];
    unattended: "supported" | "unsupported";
    readback: "supported" | "unsupported";
    close: "supported" | "unsupported";
    exactAbsence: "supported" | "unsupported";
    refundObservation: "supported" | "unsupported";
    refundCreate: "unsupported";
    authorizationCaptureSplit: "unsupported";
}
/** The neutral bundle an adapter factory returns; optional ports match declared capabilities. */
export interface PaymentAdapter {
    capabilities: PaymentAdapterCapabilities;
    attemptPort: PaymentAttemptPort;
    verifyPort: PaymentVerifyNormalizePort;
    readbackPort: PaymentReadbackPort;
    absencePort?: PaymentAbsencePort;
}
export {};

import type { PaymentMethodHandoff, PaymentRefusalEnvelope } from "./adapter.js";
/** Configured integer-minor money; currency bytes pass through unchanged. */
export type PaymentMoney = {
    amountMinor: number;
    currency: string;
};
/** Exact attempt identity for readback, close and absence lookups. */
export type PaymentAttemptIdentity = {
    provider: string;
    /** Null permits an exact intent lookup when dispatch acknowledgement was lost. */
    providerPaymentRef: string | null;
    intentRef: string;
    attemptRef: string;
};
type VerifiedIdentity = {
    provider: string;
    /** Adapter-derived; includes outcome/status class, never transaction identity alone. */
    providerEventId: string;
    /** Provider-asserted time or null; never substitute receipt time. */
    providerOccurredAt: string | null;
    /** Stable normalized evidence fingerprint; excludes receipt time and secrets. */
    fingerprint: string;
};
/** Original payment money and stable operation identity accompany every reversal. */
export type PaymentRefundObservation = {
    operationId: string;
    originalPaymentMoney: PaymentMoney;
} & ({
    kind: "cumulative-snapshot";
    refundedMoney: PaymentMoney;
} | {
    kind: "event-delta";
    refundedMoney: PaymentMoney;
} | {
    kind: "full-reversal";
    reason: "refund" | "chargeback";
});
/** Type-only policy: require-evidence is default; legacy behavior needs explicit opt-in. */
export type RefundEvidencePolicy = "require-evidence" | "legacy-assume-full";
/** Verified money facts; authorization alone is never settled money. */
export type VerifiedPaymentEvidence = VerifiedIdentity & {
    kind: "money";
    providerPaymentRef: string;
    intentRef: string | null;
    attemptRef: string | null;
    outcome: "succeeded" | "failed" | "pending" | "refunded";
    money: PaymentMoney | null;
    refusal?: PaymentRefusalEnvelope;
    refund?: PaymentRefundObservation;
};
/**
 * Method lifecycle facts, separate from the money-outcome union; setup is never money success.
 * Outcomes follow the core method lifecycle kinds plus setup failure and required customer action.
 */
export type VerifiedMethodEvidence = VerifiedIdentity & {
    kind: "method";
    outcome: "registered" | "updated" | "revoked" | "expired" | "suspended" | "failed" | "requires-action";
    methodRef: string;
    setupRef: string | null;
    intentRef: string | null;
};
/** Verification uses injected I/O and original bytes; never reserialize rawBody. */
export interface PaymentVerifyNormalizePort {
    verifyAndNormalize(input: {
        rawBody: Uint8Array;
        headers: Readonly<Record<string, string | readonly string[] | undefined>>;
        /** Freshness only; never part of identity or fingerprint. */
        receivedAt: string;
    }): Promise<VerifiedPaymentEvidence | VerifiedMethodEvidence | {
        kind: "refusal";
        reason: "signature" | "stale" | "malformed" | "unsupported-event";
    }>;
}
/** Existing purpose bytes are preserved for reference reconciliation callers. */
export type ProviderReadPurpose = "active_checkout" | "recovery";
/** Absence requires a complete bounded lookup of exact identity and configured money. */
export type PaymentAbsenceEnvelope = {
    identity: PaymentAttemptIdentity;
    expectedMoney: PaymentMoney;
} & ({
    kind: "absent";
    evidence: {
        exact: true;
        complete: true;
        lookupReference: string;
    };
} | {
    kind: "unknown";
    reason: "timeout" | "truncated" | "indeterminate";
} | {
    kind: "unsupported";
});
/** Unknown is neither absence nor unsupported. Timeout/truncation stays unknown. */
export type PaymentReadbackEnvelope = {
    identity: PaymentAttemptIdentity;
    expectedMoney: PaymentMoney;
    purpose: ProviderReadPurpose;
} & ({
    kind: "succeeded" | "failed" | "pending";
    observedMoney: PaymentMoney;
    providerOccurredAt: string | null;
    refusal?: PaymentRefusalEnvelope;
    handoff?: PaymentMethodHandoff;
} | {
    kind: "unknown";
    reason: "timeout" | "truncated" | "indeterminate";
} | {
    kind: "absent";
    evidence: Extract<PaymentAbsenceEnvelope, {
        kind: "absent";
    }>["evidence"];
} | {
    kind: "unsupported";
});
/** Exact readback of one attempt; a browser return or provider silence is never money truth. */
export interface PaymentReadbackPort {
    readPayment(input: {
        identity: PaymentAttemptIdentity;
        expectedMoney: PaymentMoney;
        purpose: ProviderReadPurpose;
    }): Promise<PaymentReadbackEnvelope>;
    /** Present only when capabilities declare close supported; returns the post-close readback. */
    close?(input: {
        identity: PaymentAttemptIdentity;
        expectedMoney: PaymentMoney;
        purpose: ProviderReadPurpose;
    }): Promise<PaymentReadbackEnvelope>;
}
/** Present only when capabilities declare exact absence supported. */
export interface PaymentAbsencePort {
    proveAbsence(input: {
        identity: PaymentAttemptIdentity;
        expectedMoney: PaymentMoney;
    }): Promise<PaymentAbsenceEnvelope>;
}
export {};

import type { PaymentAttemptStatus, PaymentExecutionBaseInput, PaymentExecutionBaseResult } from "@openlup/core/payment";
/** Durable preparation facts; provider admission belongs to the caller. */
export interface PreparedAttemptPrepareInput<P extends string = string> {
    idempotencyKey: string;
    paymentIntentId: string;
    provider: P;
    providerIdempotencyKey: string;
    providerRequestFingerprint: string;
    providerFlow: string;
    paymentMethodRef?: string | null;
    requestPayload: Record<string, unknown>;
}
export interface PreparedAttemptPersistenceResult<N extends string = string> {
    paymentAttemptId: string;
    status: PaymentAttemptStatus;
    replayed: boolean;
    providerAttemptId: string | null;
    providerSessionId: string | null;
    nextActionKind: N | null;
}
export interface PreparedAttemptFinalizeInput<N extends string = string> extends Pick<PaymentExecutionBaseResult, "providerAttemptId" | "providerSessionId" | "attemptStatus" | "requestPayload" | "responsePayload"> {
    idempotencyKey: string;
    paymentIntentId: string;
    paymentAttemptId: string;
    providerIdempotencyKey: string;
    providerRequestFingerprint: string;
    nextActionKind: N | null;
}
/** Only persistence capabilities needed by interactive attempt ordering. */
export interface PreparedAttemptPersistencePort<P extends string = string, N extends string = string> {
    prepareProviderAttempt(input: PreparedAttemptPrepareInput<P>): Promise<PreparedAttemptPersistenceResult<N>>;
    finalizeProviderAttempt(input: PreparedAttemptFinalizeInput<N>): Promise<PreparedAttemptPersistenceResult<N>>;
}
export interface PreparedAttemptExecutionPort<I extends PaymentExecutionBaseInput, R extends PaymentExecutionBaseResult> {
    validateInput?(input: I): void;
    execute(input: I): Promise<R>;
}
/** Exact execution extensions and original input/port references cross this seam. */
export interface PreparedAttemptExecutionInput<I extends PaymentExecutionBaseInput, R extends PaymentExecutionBaseResult, P extends string, N extends string> {
    paymentPort: PreparedAttemptPersistencePort<P, N>;
    executionPort: PreparedAttemptExecutionPort<I, R>;
    provider: P;
    paymentIntentId: string;
    executionInput: I;
    prepareIdempotencyKey: string;
    finalizeIdempotencyKey: string;
    providerIdempotencyKey: string;
    providerRequestFingerprint: string;
    providerFlow: string;
    paymentMethodRef?: string | null;
    prepareRequestPayload: Record<string, unknown>;
    validateExecution?: (execution: R) => void;
}
/** Synchronous, call-scoped native trust, diagnostics and error construction. */
export interface PreparedAttemptFailurePolicy<E extends Error, N extends string> {
    replayError(prepared: PreparedAttemptPersistenceResult<N>): Error;
    isTrustedNoCharge(error: unknown): error is E;
    logRetry(error: E, prepared: PreparedAttemptPersistenceResult<N>): void;
    exhaustedError(error: E, first: E, prepared: PreparedAttemptPersistenceResult<N>): E;
    executionError(error: unknown, prepared: PreparedAttemptPersistenceResult<N>): Error;
    finalizationError(error: unknown, prepared: PreparedAttemptPersistenceResult<N>): Error;
}
export interface PreparedAttemptExecutionResult<R extends PaymentExecutionBaseResult> {
    execution: R;
    attempt: {
        paymentAttemptId: string;
        status: PaymentAttemptStatus;
    };
}
