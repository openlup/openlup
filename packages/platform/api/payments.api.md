# ./payments API declaration snapshot

Maturity: **stable**.

Stable API under SemVer from 1.0.0. Unions and literal sets are open: adding a value is minor; consumers must keep a default branch for unknown values. Removing, renaming or narrowing an export, value or type is major. Symbols tagged @experimental within a stable subpath are outside this promise.

Generated by `node --experimental-strip-types ./scripts/api-contract.ts --update`. Every change requires explicit review.

```ts
export type ReconciliationProviderKind = string;
export type ProviderReconciliationStatusKind = "succeeded" | "failed" | "pending" | "unknown";
export interface ProviderReconciliationStatus {
    status: ProviderReconciliationStatusKind;
    providerStatus: string;
    occurredAt: string | null;
    failureReason: string | null;
    amountMinor: number | null;
    capturedAmountBasis?: "provider_capture_v1";

/**
 * Who is asking, and therefore what an unresolved provider state MEANS.
 *
 * The same read answers two different questions. Reconciliation — the caller
 * that names no purpose — asks "is this attempt over", long after the buyer
 * left, and an intent still waiting for its first payment method is over. A
 * buyer standing in a live checkout asks "is this attempt settled yet", and the
 * same state means the opposite: nobody has been asked anything.
 *
 * ⛔ Omitting the purpose keeps the reconciliation reading. Terminalizing real
 * abandonment is the more expensive of the two errors to lose, so it is the
 * behaviour a caller gets by default rather than one it must opt into.
 */
export interface PaymentProviderReconciliationProvider {
    readPayment(input: {
        providerPaymentId: string;
        attempt?: ClaimedPaymentAttempt;
        purpose?: ProviderReadPurpose;

    closePayment?(input: {

     * absence proof and must never authorize another charge. */
```

```ts
 * The class the reading adapter already decided, read back off the normalized
 * status so the terminal write can carry it.
 *
 * READ, never re-derived. Both shipped adapters classify what they read through
 * the shared `classifyDecline` seam and put the verdict in `rawPayload`; deriving
 * a second one here would be a second classifier, free to disagree with the one

 * consumers of that contract are pinned to.
```

```ts
/** Synthetic transport outcomes; no provider SDK, network, timer or retry. */
export type ScriptedPaymentOutcome<T> = {
    kind: "success";
    value: T;
} | {
    kind: "provider-5xx";
    response?: T;
} | {
    kind: "timeout-before-dispatch";
} | {
    kind: "timeout-after-dispatch";
} | {
    kind: "redelivery-409";
    response?: T;
};
export declare class ScriptedPaymentTransportError extends Error {
    readonly kind: Exclude<ScriptedPaymentOutcome<never>["kind"], "success">;
    readonly status: number | undefined;
    readonly dispatched: boolean;
    constructor(kind: Exclude<ScriptedPaymentOutcome<never>["kind"], "success">);
}
/** Per-call FIFO extracted from adapter test clients. Arguments/secrets are never recorded.
 * HTTP bindings may supply a response for 5xx/409; SDK bindings receive an error.
 * Dispatch observations describe the script only, never production no-charge evidence.
 */
export declare function createScriptedPaymentFake<Calls extends Record<string, unknown>>(): {
    calls: readonly {
        call: keyof Calls;
        kind: ScriptedPaymentOutcome<unknown>["kind"];
    }[];
    enqueue<K extends keyof Calls>(call: K, ...outcomes: ScriptedPaymentOutcome<Calls[K]>[]): void;
    invoke<K extends keyof Calls>(call: K): Promise<Calls[K]>;
    assertDrained(): void;
};
```

```ts
import type { PaymentIngressPort, PaymentVerifyNormalizePort, RefundEvidencePolicy, VerifiedMethodEvidence, VerifiedPaymentEvidence } from "./contracts/evidence.js";
declare const PAYMENT_WEBHOOK_EVENT_TYPES: readonly ["payment.succeeded", "payment.failed", "payment.refunded", "payment.requires_action", "payment.disputed", "setup.succeeded", "setup.failed", "setup.requires_action"];
type PaymentWebhookEventType = (typeof PAYMENT_WEBHOOK_EVENT_TYPES)[number];
export type ProviderPaymentWebhookKind = PaymentWebhookEventType;
export interface NormalizedProviderPaymentWebhook {
    provider: string;
    providerEventId: string;
    eventType: ProviderPaymentWebhookKind;
    providerPaymentId: string;
    paymentIntentId?: string | null;
    paymentAttemptId?: string | null;
    amountMinor?: number | null;
    currency?: string | null;
    capturedAmountBasis?: VerifiedPaymentEvidence["capturedAmountBasis"];
    occurredAt: string;
    refundKind?: "full" | "partial" | null;
    rawPayload: Record<string, unknown>;
    reusableMethod?: {
        clientId: string;
        subscriptionId?: string | null;
        providerCustomerRef?: string | null;
        providerMethodRef: string;
        providerMandateRef?: string | null;
        methodKind: string;
        status: "pending_verification" | "active" | "inactive" | "expired" | "revoked";
        consentSnapshot?: Record<string, unknown>;
        expiresAt?: string | null;
    } | null;
}
export interface PaymentWebhookControlPort {
    findCompletedHistoricalCapture?(input: {
        provider: string;
        providerEventId: string;
        providerPaymentId: string;
        legacyFingerprint: string;
    }): Promise<string | null>;
    ingestPaymentEvent(input: NormalizedProviderPaymentWebhook): Promise<{
        paymentEventId: string;
        paymentIntentId: string | null;
        paymentAttemptId: string | null;
        replayed: boolean;
        occurredAt?: string;
    }>;
    applyPaymentResult?(input: {
        idempotencyKey: string;
        paymentIntentId: string;
        paymentEventId: string;
        resultStatus: "succeeded" | "failed" | "refunded" | "partially_refunded" | "disputed";
        occurredAt: string;
        failureReason: string | null;
    }): Promise<{
        replayed: boolean;
        kind?: string;
    }>;
    confirmSubscriptionActivation?(input: {
        idempotencyKey: string;
        paymentIntentId: string;
        occurredAt: string;
        methodRef: string | null;
        methodKind: string | null;
    }): Promise<{
        confirmed: boolean;
        status: string | null;
    }>;
    recoverPaidSubscriptionActivationWithCard?(input: {
        idempotencyKey: string;
        subscriptionId: string;
        paymentMethodRef: string;
        occurredAt: string;

        recovered: boolean;
        replayed: boolean;
        reason: string | null;
    }>;

        paymentEventId: string;
    }): Promise<{
        updated: boolean;
    }>;

export interface PaymentWebhookMethodRefPort {
    upsertFromWebhook(input: NormalizedProviderPaymentWebhook): Promise<{
        replayed: boolean;
    }>;
}

        paymentAttemptId: string | null;

export interface PaymentWebhookProcessingPorts {

        clientId: string;
```
