/** PostgREST-shaped error the consumers narrow on (`.code`, `.message`, `.details`, `.hint`). */
export interface PgShimError {
    code?: string;
    message?: string;
    details?: string;
    hint?: string;
}
export interface PgShimResult<T = unknown> {
    data: T;
    error: PgShimError | null;
    count?: number | null;
}
/** Minimal query surface we need from a pg client (Pool acquires a PoolClient per request). */
export interface PgQueryExecutor {
    query(text: string, values?: unknown[]): Promise<{
        rows: Record<string, unknown>[];
    }>;
}
export type Filter = {
    kind: "eq" | "gte" | "lte" | "gt";
    column: string;
    value: unknown;
} | {
    kind: "in";
    column: string;
    values: unknown[];
} | {
    kind: "ilike";
    column: string;
    value: string;
} | {
    kind: "is";
    column: string;
    value: unknown;
};
export interface OrderBy {
    column: string;
    ascending: boolean;
}
export declare function quoteIdent(raw: string): string;
export declare function renderColumns(columns: string): string;
export declare function renderFilter(filter: Filter, params: ParamAccumulator): string;
/**
 * PostgREST `count=exact` rendered as a SEPARATE aggregate over the same FROM/WHERE — deliberately
 * NOT a `COUNT(*) OVER()` folded into the row query. PostgREST's exact count ignores limit/offset,
 * and a window function emits no row at all for an empty page, so an offset past the end would
 * report 0 instead of the true total. Keeping it separate also leaves the row projection
 * byte-identical: there is no synthetic column to strip before cardinality shaping.
 */
export declare function renderCountQuery(table: string, filters: readonly Filter[], params: ParamAccumulator): string;
/**
 * PostgREST refuses an out-of-bounds page instead of answering an empty one -- but ONLY when it
 * knows the total, i.e. when an exact count was requested; without one it answers 200 and `[]`,
 * which is what this shim already did. The boundary is strict: offset === total is a legal empty
 * page, offset === total + 1 is not. Measured against PostgREST 14.10 before this was written,
 * including the `details` sentence, which carries the two numbers a caller pages on.
 */
export declare function rangeNotSatisfiable(offset: number | null, total: number | null): PgShimError | null;
/**
 * INSERT ... VALUES ... RETURNING, with the column list taken from the FIRST row and every later
 * row required to carry the same key set. Deriving the columns from row 0 alone and binding the
 * rest positionally is how a batch used to lose a column silently: an extra key vanished, a missing
 * one bound NULL over a DEFAULT. PostgREST answers PGRST102 "All object keys must match" to exactly
 * this payload -- and keeps answering it under `Prefer: missing=default`, measured -- so the shim
 * refuses too. Refusal, not union: parity is the point, and this builder exposes no Prefer header
 * through which a caller could ask for anything else.
 */
export declare function renderInsertStatement(table: string, rows: readonly Record<string, unknown>[], projection: string, params: ParamAccumulator): string;
export declare function unwrapRpcRows(rows: Record<string, unknown>[], functionName: string): unknown;
/** Accumulates values and hands back $1,$2,… placeholders — the single injection-safety chokepoint. */
export declare class ParamAccumulator {
    readonly values: unknown[];
    add(value: unknown): string;
}
export declare function joinParts(parts: string[]): string;
export declare function toShimError(error: unknown): PgShimError;

/** Structural M03 contributor; identity owns authorization, transactions and receipt replay. */
export interface RiskPrivacyContributor<Context> {
    readonly id: string;
    readonly version: string;
    readonly ownedResources: readonly string[];
    prepareErasure(context: Context): Promise<void>;
    export(context: Context, subject: RiskPrivacySubject): Promise<PersonalDataDocument>;
    assessErasure(context: Context, subject: RiskPrivacySubject): Promise<RiskPrivacyAssessment>;
    eraseOrRetain(context: Context, subject: RiskPrivacySubject, assessment: RiskPrivacyAssessment, actor: {
        principalId: string;
    }): Promise<{
        contributorId: string;
        version: string;
        obligations: PersonalDataObligation[];
    }>;
}
/** Logical resources cover both rails, including embedded native replay fingerprints.
 * Review cases include hold links; commerce still owns the linked hold itself.
 */
export declare const RISK_PRIVACY_RESOURCES: readonly ["risk.assessments", "risk.blocklist-entries", "risk.review-cases", "risk.case-events", "risk.idempotency"];
export type RiskPrivacyResource = typeof RISK_PRIVACY_RESOURCES[number];
export type RiskPrivacyReadResult = {
    resource: RiskPrivacyResource;
    status: "available" | "unavailable";
    /** Explicitly selected subject data only; unavailable resources must use null. */
    data: PrivacyJson;
};
export interface RiskPrivacyReadPort<Context> {
    /** Read-only in the identity transaction. Map resolved IDs to rail-specific order,
     * client, actor and hashed subject links; never infer ownership by equal emails.
     * Return exactly one result per logical resource, even if empty on this rail.
     * Omit secrets/credentials. Missing grants or uncertain linkage mean unavailable.
     */
    read(context: Context, subject: RiskPrivacySubject): Promise<readonly RiskPrivacyReadResult[]>;
}
export interface RiskPrivacyPolicy<Context> {
    /** Read-only, explicit owner/resource obligations for this subject and deployment.
     * Only retain or unresolved is supported. No default retention period or deletion.
     * Policy prose is not copied to exports/receipts; reasons are projected to fixed codes.
     */
    assess(context: Context, subject: RiskPrivacySubject): Promise<readonly PersonalDataObligation[]>;
}
export interface RiskPrivacyDependencies<Context> {
    data: RiskPrivacyReadPort<Context> | null;
    policy: RiskPrivacyPolicy<Context> | null;
}
/** Bounded read/retention contributor: never deletes audit, replay, hash or hold facts. */
export declare function createRiskPrivacyContributor<Context>(input: RiskPrivacyDependencies<Context>): RiskPrivacyContributor<Context>;
/** Opaque host-owned future binding. This selects no transfer/restore contract or runtime. */
export interface RiskPrivacyBinding<Context, Portability> {
    contributor: RiskPrivacyContributor<Context>;
    portability: {
        status: "deferred";
    } | {
        status: "host-bound";
        binding: Portability;
    };
}
/** Append contributor to the existing M03 registry; no default/reference registration.
 * A supplied portability value is carried unchanged, never invoked or certified.
 */
export declare function createRiskPrivacyBinding<Context, Portability = never>(input: RiskPrivacyDependencies<Context> & {
    portability?: Portability;
}): RiskPrivacyBinding<Context, Portability>;
