```ts
/** Exact UTF-8 scaffold files; every unlisted path remains application-owned. */
export interface AppVersionSnapshot {
    version: string;
    schema: {
        min: number;
        max: number;
    };
    files: Readonly<Record<string, string>>;
}
export interface AppUpgradeInput {
    /** Absolute canonical path (as returned by realpath) of an existing app directory. */
    root: string;
    from: AppVersionSnapshot;
    to: AppVersionSnapshot;
    /** Observed by the caller; planning and applying execute no database work. */
    schemaVersion: number;
}
export type AppUpgradeAction = "create" | "replace" | "remove" | "unchanged" | "preserved" | "already-applied";
export interface AppUpgradeConflict {
    path: string;
    reason: string;
}
export interface AppUpgradePlan {
    fromVersion: string;
    toVersion: string;
    schemaVersion: number;
    files: {
        path: string;
        action: AppUpgradeAction;
    }[];
    conflicts: AppUpgradeConflict[];
}
export declare class AppUpgradeConflictError extends Error {
    readonly conflicts: readonly AppUpgradeConflict[];
    constructor(conflicts: readonly AppUpgradeConflict[]);
}
/** Read-only comparison of explicit predecessor and candidate snapshots. */
export declare function planAppUpgrade(input: AppUpgradeInput): Promise<AppUpgradePlan>;
/**
 * The caller must hold an exclusive maintenance window over the application tree.
 * This is not a lock against concurrent editors or hostile filesystem mutation.
 * All conflicts are checked before any writes. A later I/O failure can leave a
 * partial upgrade: rerun the same snapshots to recognize exact completed bytes.
 * Each replacement is atomic on a local filesystem; the complete upgrade is not.
 * An abrupt process termination can leave a uniquely named temporary file; it is
 * never treated as a managed app file or resumed operation.
 * No SQL, dependency install, version marker, deployment or user hook is executed.
 */
export declare function applyAppUpgrade(input: AppUpgradeInput): Promise<{
    plan: AppUpgradePlan;
    applied: string[];
}>;
```

```ts
/** Read-only prerequisites of the immutable legacy fresh-install bootstrap, not install readiness. */
export interface PostgresInstallQuery {
    query(sql: string, values?: unknown[]): Promise<{
        rows: Record<string, unknown>[];
    }>;
}
export interface LegacyPostgresInstallRequest {
    /** Expected actual login: both current_user and session_user must match. */
    migrationLogin: string;
    /** Legacy bootstrap requires this to be the same login as the migrator. */
    runtimeLogin: string;
    roleMode: "installer-managed" | "operator-provisioned";
}
export interface PostgresInstallRoleObservation {
    name: string;
    login: boolean;
    superuser: boolean;
    inherit: boolean;
    createRole: boolean;
    createDatabase: boolean;
    replication: boolean;
    bypassRls: boolean;
}
export interface PostgresInstallMembershipObservation {
    role: string;
    member: string;
    grantor: string;
    admin: boolean;
    inherit: boolean;
    set: boolean;
}
export type LegacyPostgresInstallBlocker = "migration_login_mismatch" | "legacy_bootstrap_requires_superuser" | "legacy_bootstrap_requires_shared_runtime_login" | "runtime_login_missing_or_not_login" | "legacy_bootstrap_requires_installer_managed_roles" | "legacy_bootstrap_role_already_exists";
export interface LegacyPostgresInstallRoleReport {
    scope: "legacy-fresh-bootstrap-role-prerequisites";
    database: string;
    currentUser: string;
    sessionUser: string;
    roles: PostgresInstallRoleObservation[];
    memberships: PostgresInstallMembershipObservation[];
    blockers: {
        code: LegacyPostgresInstallBlocker;
        role?: string;
    }[];
    rolePrerequisitesMet: boolean;
}
/**
 * Observes PostgreSQL 16+ legacy role prerequisites without changing the database.
 * A true result says nothing about schema inventory, object ACLs, pooling, import safety or hosting.
 * Query/driver errors are redacted because they can contain connection credentials.
 */
export declare function inspectLegacyPostgresInstallRoles(db: PostgresInstallQuery, request: LegacyPostgresInstallRequest): Promise<LegacyPostgresInstallRoleReport>;
```
