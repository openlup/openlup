## dist/portability/objects.d.ts

```ts
export type ObjectTransferFailure = "absent" | "denied" | "unavailable" | "version_conflict" | "integrity_conflict" | "invalid_input";
/** Adapters distinguish missing data from permission and availability failures. */
export declare class ObjectTransferError extends Error {
    readonly code: ObjectTransferFailure;
    constructor(code: ObjectTransferFailure);
}
export interface TransferObject {
    key: string;
    /** Logical immutable source version; never a claim about a target provider's native version ID. */
    version: string;
    bytes: number;
    sha256: string;
    metadata: Readonly<Record<string, string>>;
    /** An explicit access classification, never inferred from a public URL. */
    access: string;
    lineage?: {
        sourceId: string;
        snapshotId: string;
        sourceVersion: string;
    };
}
export interface TransferObjectRead {
    object: TransferObject;
    body: AsyncIterable<Uint8Array>;
}
export interface TransferObjectReader {
    /** Must read the requested version or refuse, never silently read a newer one. */
    read(key: string, expectedVersion?: string): Promise<TransferObjectRead>;
}
export interface TransferObjectStore extends TransferObjectReader {
    /** Atomic publication; an existing key must refuse with version_conflict. */
    createIfAbsent(object: TransferObject, bytes: Uint8Array): Promise<void>;
}
export interface TransferObjectInventory {
    sourceId: string;
    /** Host prerequisite: complete inventory and stable versioned reads, or a held write fence. */
    snapshot: {
        kind: "immutable" | "quiesced";
        id: string;
    };
    objects: readonly TransferObject[];
}
/** Canonical descriptor encoding also preserves metadata independent of property order. */
export declare function encodeTransferObject(object: TransferObject): string;
export declare function validateTransferObject(object: TransferObject): void;
/** Bounded in-memory object copy; ceilings are explicit and checked before any destination mutation. */
export declare function copyTransferObjects(input: {
    inventory: TransferObjectInventory;
    source: TransferObjectReader;
    destination: TransferObjectStore;
    accessMapping: Readonly<Record<string, string>>;
    maximumObjectBytes: number;
    maximumTotalBytes: number;
    /** Called only after destination descriptor and bytes have been read back successfully. */
    checkpoint?: (object: TransferObject) => Promise<void>;
}): Promise<{
    copied: number;
    replayed: number;
    verifiedKeys: string[];
}>;
/** Read exactly the declared bytes and hash; never trust a successful HEAD alone. */
export declare function verifyRead(read: TransferObjectRead, expected: TransferObject): Promise<Uint8Array>;
export declare function normalizeObjectTransferError(error: unknown): ObjectTransferError;
```

/** A finite compatible-row utility, not a complete-store migration profile. */
declare const TABLES: readonly ["clients", "subscriptions", "subscription_events", "subscription_pause_windows", "subscription_renewal_terms", "subscription_renewal_term_lines", "subscription_renewal_operations", "subscription_cycles", "commerce_orders", "commerce_order_items", "commerce_settlement_intents", "commerce_settlement_transitions", "payment_truth_events", "payment_truth_reconciliation_evidence", "subscription_renewal_outcomes", "outbox_events", "invoicing_duties", "invoicing_fact_revisions", "invoicing_effect_receipts", "invoicing_capture_faults"];
export type CompatiblePostgresTable = typeof TABLES[number];
export interface TransferPostgresQuery {
    query<Row extends Record<string, unknown> = Record<string, unknown>>(sql: string, values?: readonly unknown[]): Promise<{
        rows: Row[];
    }>;
}
export interface CompatiblePostgresSnapshot {
    /** Own one read-only, repeatable-read transaction for the complete callback. */
    withSnapshot<T>(work: (db: TransferPostgresQuery) => Promise<T>): Promise<T>;
}
export interface CompatiblePostgresTarget {
    /** Own one transaction; commit only after the complete callback succeeds. */
    withTransaction<T>(work: (db: TransferPostgresQuery) => Promise<T>): Promise<T>;
}
type Column = {
    name: string;
    type: string;
    key: boolean;
};
type TableRows = {
    table: CompatiblePostgresTable;
    columns: Column[];
    rows: string[];
};
export interface CompatiblePostgresPacket {
    format: "openlup.compatible-rows.v1";
    sourceId: string;
    /** Opaque selected-catalogue identity supplied by the validated host capability. */
    catalogueId: string;
    tables: TableRows[];
    sha256: string;
}
export type PostgresTransferFailure = "invalid_input" | "unsupported_table" | "schema_conflict" | "insert_trigger_requires_owner" | "target_conflict" | "limit_exceeded" | "database_unavailable";
export declare class PostgresTransferError extends Error {
    readonly code: PostgresTransferFailure;
    readonly table?: string | undefined;
    constructor(code: PostgresTransferFailure, table?: string | undefined);
}
/** Data only: no schema, role, grant, migration ledger, control or provider-internal copy. */
export declare function captureCompatiblePostgresRows(input: {
    source: CompatiblePostgresSnapshot;
    sourceId: string;
    catalogueId: string;
    tables: readonly CompatiblePostgresTable[];
    maximumRows: number;
    maximumBytes: number;
}): Promise<CompatiblePostgresPacket>;
/**
 * Atomic insert and exact replay. A differing row refuses; there is no upsert,
 * deletion scan, status reset or permission change. Existing parent rows remain
 * prerequisites, and this utility never authorizes target effect execution.
 */
export declare function restoreCompatiblePostgresRows(input: {
    target: CompatiblePostgresTarget;
    catalogueId: string;
    packet: CompatiblePostgresPacket;
}): Promise<{
    inserted: number;
    replayed: number;
    sha256: string;
}>;
export {};
