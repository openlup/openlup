```ts
import type { ClientEventReporter, SubmitFailureClassifier, SubmitTransportBinding } from "./bindings.js";
/**
 * What `recoverAmbiguousSubmit` learned about a failed submit.
 *
 * - `recovered`: the one replay succeeded; `response` is the transport's.
 * - `no_stable_identity`: the submit may have committed, but there is no
 *   retained request or it carries no command key, so nothing was sent.
 * - `in_flight`: the server refused because an attempt for this purchase is
 *   already in flight, on the first submit or on the replay.
 * - `unknown`: the replay failed for any other reason; nothing more was sent.
 */
export type AmbiguousSubmitRecovery<TResponse> = {
    kind: "recovered";
    response: TResponse;
} | {
    kind: "no_stable_identity";
} | {
    kind: "in_flight";
} | {
    kind: "unknown";
};
/**
 * Decides what a failed submit means and replays it, at most once, when the
 * server may have committed it.
 *
 * It calls `classifySubmitFailure` once on `error`:
 *
 * - `attempt_in_flight`: calls `report("submit_refused_in_flight")` once when
 *   `report` is bound, sends nothing and returns `in_flight`, with or without a
 *   request.
 * - `timeout` or `transport_uncertain`: with no `request` (`null`, or
 *   `undefined` at run time), returns `no_stable_identity` without calling
 *   `transport.commandKey`. When `commandKey(request)` is falsy, returns
 *   `no_stable_identity` and sends nothing. Otherwise calls `transport.replay`
 *   once with the same `request` object, unmodified, the caller's `options`
 *   and a deadline of 10 000 ms, and returns `recovered` with its response.
 *   When the replay fails, it classifies that error once and returns
 *   `in_flight` for `attempt_in_flight` and `unknown` for any other class,
 *   reporting nothing and sending nothing more.
 * - `other`, or a class this version does not know: returns `null` and sends
 *   nothing. An unrecognized class is never replayed.
 *
 * The package passes `deadlineMs` to the transport; enforcing it is the
 * transport's job. An error that the classifier, `commandKey` or the reporter
 * raises propagates unchanged.
 */
export declare function recoverAmbiguousSubmit<TRequest, TOptions, TResponse>(bindings: {
    transport: SubmitTransportBinding<TRequest, TOptions, TResponse>;
    classifySubmitFailure: SubmitFailureClassifier;
    /**
     * Receives the in-flight refusal of the first submit, if bound.
     *
     * @experimental
     */
    report?: ClientEventReporter;
}, input: {
    error: unknown;
    request: TRequest | null;
    options: TOptions;
}): Promise<AmbiguousSubmitRecovery<TResponse> | null>;
```

## dist/storefront/bindings.d.ts

```ts
import type { PaymentAttemptStatus } from "@openlup/core/payment";
import type { AttemptNumber, CommandKey, ContinuationId } from "./identity.js";
/**
 * What a store's client keeps for one purchase so that it survives a lost
 * response and a reload.
 *
 * The record's identity is its continuation identity. `fingerprint` is the
 * host's opaque stamp of the payload that first created the record. `attempt`
 * absent means the first attempt, `0`. The record carries no time.
 */
export interface ContinuationRecord {
    continuationId: ContinuationId;
    fingerprint: string;
    attempt?: AttemptNumber;
}
/**
 * The host's storage for the continuation record. The host owns the storage
 * name and the serialization; a read returns `null` for an absent or unreadable
 * record.
 *
 * It is synchronous, matching the synchronous browser storage it is bound to
 * and the functions that read it. An asynchronous variant would be a new name.
 * A read does not filter by identity format: the identifier binding's
 * `accepts` is applied only when an identity would be reused.
 */
export interface ContinuationRecordStorageBinding {
    read(): ContinuationRecord | null;
    write(record: ContinuationRecord): void;
    clear(): void;
}
/**
 * The host's source of continuation identities.
 *
 * `next` mints a new identifier and `accepts` decides whether a stored one may
 * be reused. Both use a plain `string`: the identifier format belongs to the
 * host, and the package brands a value with `toContinuationId` only once the
 * host accepts it. A host that wants the server's format applies the commerce
 * entry's `checkoutContinuationJourneyIdSchema` in `accepts`. `accepts` is
 * applied only when a stored identity would be reused; reading or advancing a
 * stored record does not filter it by identity format.
 */
export interface IdentifierBinding {
    next(): string;
    accepts(value: string): boolean;
}
/**
 * The class the host gives a failed submit.
 *
 * - `attempt_in_flight`: the server refused because an attempt for this
 *   purchase is already in flight. Report the conflict and send nothing.
 * - `timeout`: the request timed out, so the server may have committed. One
 *   exact replay of the retained request.
 * - `transport_uncertain`: the browser reported a transport failure, so the
 *   server may have committed. One exact replay of the retained request.
 * - `other`: anything unrecognized. Send nothing and never replay.
 *
 * The union is open: members may be added in a minor release. The package reads
 * this value and treats a member it does not know as `other`.
 */
export type SubmitFailureClass = "attempt_in_flight" | "timeout" | "transport_uncertain" | "other";
/** Maps a failed submit's error to one `SubmitFailureClass`; returns `other` for anything it does not recognize. */
export type SubmitFailureClassifier = (error: unknown) => SubmitFailureClass;
/**
 * The host's submit transport, used to replay a retained request exactly.
 *
 * `commandKey` returns the request's `CommandKey`, or `null` or `undefined`
 * when the request carries none; a request without a key is never replayed.
 * `replay` sends the retained request and options again. `deadlineMs` is a
 * duration in milliseconds that the package passes to the transport; the
 * package does not enforce it, and enforcing it is the transport's job.
 */
export interface SubmitTransportBinding<TRequest = unknown, TOptions = unknown, TResponse = unknown> {
    commandKey(request: TRequest): CommandKey | null | undefined;
    replay(request: TRequest, options: TOptions, deadlineMs: number): Promise<TResponse>;
}
/**
 * What the client does when it cannot persist a new continuation identity.
 *
 * - `refuse`: the default. Do not submit.
 * - `continue_unpersisted`: deprecated from its first release. Submit with an
 *   identity that was not persisted, so a reload cannot find it again.
 *
 * The union is open: members may be added in a minor release. The package reads
 * a member it does not know as `refuse`.
 *
 * No function reads this policy yet: `getOrCreateContinuationId` continues
 * with an identity that was not persisted when its storage binding drops a
 * write. A later release is planned to add the storage-failure policy
 * parameter (default `refuse`) as a declared behaviour change.
 *
 * @experimental
 */
export type StorageFailurePolicy = "refuse" | "continue_unpersisted";
/**
 * Receives a client event the package reports.
 *
 * New events may be added in a minor release, so a host accepts the parameter's
 * union type rather than narrowing to the literals it knows.
 *
 * @experimental
 */
export type ClientEventReporter = (event: "submit_refused_in_flight") => void;
/**
 * Marks a provider action that was already issued for a purchase.
 *
 * `actionKind` is the kind of action, `phase` how far the client got with it,
 * and `expiresAt` the time after which the marker no longer describes a live
 * action, in epoch seconds. A marker without `phase` is read as
 * `confirm_dispatched`; a marker without `expiresAt` does not expire.
 *
 * Both unions are open: members may be added in a minor release. The package
 * reads an `actionKind` it does not know as `none`, so it reopens or follows no
 * action, and a `phase` it does not know as `confirm_dispatched`, so it keeps
 * waiting on a charge that may be live.
 *
 * @experimental
 */
export interface PendingAction {
    actionKind: "embedded" | "redirect" | "none";
    phase?: "action_issued" | "confirm_dispatched";
    expiresAt?: number;
}
/**
 * The host's storage for the pending-action marker. The host owns the storage
 * name and the serialization, and may extend the marker with its own fields
 * through `C`.
 *
 * @experimental
 */
export interface PendingActionStorageBinding<C extends PendingAction = PendingAction> {
    read(): C | null;
    write(pending: C): void;
    clear(): void;
}
/**
 * What a status binding resolves to.
 *
 * - `terminal`: the server answered that the purchase is over.
 * - `checking`: the status could not be read. It is not an answer.
 * - `active`: the purchase is open, with optional payment and action facts.
 *
 * The provider fields are opaque strings the host supplies; the package names
 * no provider. A host maps a payment attempt status it does not recognize to
 * `null`, which the package reads as "not known to admit a new attempt".
 *
 * The union and the `action.kind` union are open: members may be added in a
 * minor release. The package reads a reading kind it does not know as
 * `checking`, and an action kind it does not know as `other`, which it does
 * not act on.
 *
 * @experimental
 */
export type StatusReading = {
    kind: "terminal";
} | {
    kind: "checking";
} | {
    kind: "active";
    payment?: {
        provider?: string | null;
        attemptStatus?: PaymentAttemptStatus | null;
    } | null;
    action?: {
        kind: "embedded";
        provider?: string;
        actionToken?: string;
    } | {
        kind: "redirect";
        url: string;
    } | {
        kind: "other";
        provider?: string;
    } | null;
};
/**
 * The host's status read for a pending action.
 *
 * It resolves to `checking` when the status cannot be read, and to `terminal`
 * only on the server's answer that the purchase is over.
 *
 * @experimental
 */
export interface StatusTransportBinding<C extends PendingAction = PendingAction> {
    read(pending: C): Promise<StatusReading>;
}
/**
 * Facts about the payment providers a store uses, as opaque names.
 *
 * `providers` lists the providers, `reopenableEmbeddedProviders` those whose
 * embedded action can be opened again, and `retryableAttemptStatuses` the
 * attempt statuses that admit a new attempt. The default for the last is the
 * core payment entry's `RETRYABLE_ATTEMPT_STATUSES`; the statuses are core's
 * `PaymentAttemptStatus`, so this entry adds no second vocabulary.
 *
 * @experimental
 */
export interface PaymentProviderFacts {
    providers: readonly string[];
    reopenableEmbeddedProviders: readonly string[];
    retryableAttemptStatuses: readonly PaymentAttemptStatus[];
}
/**
 * What the server does with a changed payload sent under the same command key:
 * `update_in_place` updates the existing record, `refuse_changed` refuses it.
 * It configures no replay. The union is open: members may be added in a minor
 * release. The package reads a member it does not know as `refuse_changed`: it
 * never assumes that the server applied a changed payload.
 *
 * @experimental
 */
export type ChangedPayloadPolicy = "update_in_place" | "refuse_changed";
/**
 * The class of the result of any submit command, not only a purchase.
 *
 * The union is open: members may be added in a minor release. The package reads
 * this value and treats a member it does not know as `unknown`.
 *
 * @experimental
 */
export type SubmitOutcomeClass = "settled" | "needs_client_action" | "awaiting_authority" | "refused" | "unknown";
/**
 * Maps a submit command's result to one `SubmitOutcomeClass`.
 *
 * @experimental
 */
export type SubmitOutcomeClassifier<TResult = unknown> = (result: TResult) => SubmitOutcomeClass;
```
## dist/storefront/continuationRecord.d.ts

```ts
import type { ContinuationRecord, ContinuationRecordStorageBinding, IdentifierBinding } from "./bindings.js";
import { type ContinuationId } from "./identity.js";
/**
 * Returns the continuation identity of the current purchase.
 *
 * It reuses the stored record's identity when `identifier.accepts` accepts it;
 * reuse writes nothing and does not call `fingerprint`. Otherwise it calls
 * `identifier.next()`, then `fingerprint()`, writes a new record with that
 * fingerprint and identity and no attempt, replacing any stored record, and
 * returns the new identity. One purchase therefore keeps one identity across
 * payload changes until `clearContinuationRecord` runs.
 *
 * It reads no storage-failure policy. A storage binding that cannot reach its
 * storage returns `null` from `read` and drops `write`, so each call returns a
 * new identity that is not kept. An error a binding or `fingerprint` raises
 * propagates unchanged.
 */
export declare function getOrCreateContinuationId(bindings: {
    storage: ContinuationRecordStorageBinding;
    identifier: IdentifierBinding;
}, input: {
    fingerprint: () => string;
}): ContinuationId;
/**
 * Returns the stored continuation record as the storage binding reads it, or
 * `null`. It applies no acceptance check and writes nothing, so it also returns
 * a record whose identity `getOrCreateContinuationId` would not reuse. An
 * absent `attempt` means the first attempt, `0`.
 */
export declare function readContinuationRecord(bindings: {
    storage: ContinuationRecordStorageBinding;
}): ContinuationRecord | null;
/**
 * Moves the stored record to its next payment attempt: it writes the record
 * with `attempt` one higher, reading an absent `attempt` as `0`, and keeps its
 * identity and fingerprint whatever the identity's format. With no stored
 * record it writes nothing. It is the only function that changes `attempt`.
 */
export declare function advanceAttempt(bindings: {
    storage: ContinuationRecordStorageBinding;
}): void;
/**
 * Clears the stored continuation record, so the next `getOrCreateContinuationId`
 * mints a new identity.
 */
export declare function clearContinuationRecord(bindings: {
    storage: ContinuationRecordStorageBinding;
}): void;
```
## dist/storefront/identity.d.ts

```ts
declare const continuationIdBrand: unique symbol;
declare const commandKeyBrand: unique symbol;
declare const attemptNumberBrand: unique symbol;
/**
 * The identity of one buyer purchase and of its continuation record.
 *
 * It is the browser-side opaque handle for the server's journey identifier,
 * whose format the commerce entry's `checkoutContinuationJourneyIdSchema`
 * validates. This package does not validate it: a host that wants validation
 * applies that schema in its `IdentifierBinding.accepts`. A plain `string`, a
 * `CommandKey` and an `AttemptNumber` are not assignable to it.
 */
export type ContinuationId = string & {
    readonly [continuationIdBrand]: true;
};
/**
 * The replay identity of one command: a request sent again under the same key
 * is the same command. A plain `string` and a `ContinuationId` are not
 * assignable to it.
 */
export type CommandKey = string & {
    readonly [commandKeyBrand]: true;
};
/**
 * Counts payment attempts within one purchase. The first stored attempt is `0`.
 * A plain `number` is not assignable to it.
 */
export type AttemptNumber = number & {
    readonly [attemptNumberBrand]: true;
};
/**
 * Brands a value as a continuation identity. Returns the value unchanged and
 * never throws, so a host's codec can brand any stored value its own
 * acceptance check accepts.
 */
export declare function toContinuationId(value: string): ContinuationId;
/** Brands a value as a command key. Returns the value unchanged and never throws. */
export declare function toCommandKey(value: string): CommandKey;
/** Brands a value as an attempt number. Returns the value unchanged and never throws. */
export declare function toAttemptNumber(value: number): AttemptNumber;
export {};
```
